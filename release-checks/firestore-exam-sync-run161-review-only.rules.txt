// Match specifically user-owned exam progress, not public users or sync_v3.
match /mock_exam_progress_v1/{uid} {
  function examOwner() {
    return request.auth != null &&
      request.auth.uid == uid &&
      request.auth.token.email_verified == true;
  }
  function validExamSnapshot() {
    let d = request.resource.data;
    return d.keys().hasOnly(['schema','ownerUid','attemptId','updatedAt','answersCount','position','snapshot']) &&
      d.keys().hasAll(['schema','ownerUid','attemptId','updatedAt','answersCount','position','snapshot']) &&
      d.schema == 1 &&
      d.ownerUid == uid &&
      d.attemptId is string &&
      d.attemptId.matches('^A_[a-f0-9]+_[a-f0-9]+$') &&
      d.updatedAt is int && d.updatedAt > 0 &&
      d.answersCount is int && d.answersCount >= 0 && d.answersCount <= 40 &&
      d.position is int && d.position >= 0 && d.position <= 40 &&
      d.snapshot is string && d.snapshot.size() > 0 && d.snapshot.size() <= 300000;
  }
  allow get: if examOwner();
  allow list: if false;
  allow create, update: if examOwner() && validExamSnapshot();
  allow delete: if false;
}
